cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(fuota_storage_lifecycle)
get_filename_component(R "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
target_include_directories(app PRIVATE ${R}/src/lorawan ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_sources(app PRIVATE src/main.c)
get_property(LIBS GLOBAL PROPERTY ZEPHYR_LIBS)
foreach(LIB ${LIBS})
  get_target_property(SOURCES ${LIB} SOURCES)
  foreach(SOURCE ${SOURCES})
    if(SOURCE MATCHES "services/fuota/lorawan_fuota.c$")
      target_include_directories(${LIB} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
      set_source_files_properties(${SOURCE} TARGET_DIRECTORY ${LIB} PROPERTIES COMPILE_DEFINITIONS "CONFIG_RZI_LORAWAN_BACKEND_USP=1" COMPILE_OPTIONS "-include;${CMAKE_CURRENT_SOURCE_DIR}/include/inject.h")
    endif()
  endforeach()
endforeach()
